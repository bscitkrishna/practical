using System;
using System.Drawing;

public partial class _Default : System.Web.UI.Page
{
    protected void Red_Click(object sender, EventArgs e)
    {
        Label1.CssClass = "red";
    }

    protected void Green_Click(object sender, EventArgs e)
    {
        Label1.CssClass = "green";
    }

    protected void Blue_Click(object sender, EventArgs e)
    {
        Label1.CssClass = "blue";
    }

    protected void ChangeFont(object sender, EventArgs e)
    {
        Label1.Font.Name = font.SelectedValue;
    }

    protected void ChangeSize(object sender, EventArgs e)
    {
        Label1.Font.Size =
            FontUnit.Point(int.Parse(size.SelectedValue));
    }

    protected void ChangeStyle(object sender, EventArgs e)
    {
        Label1.Font.Bold = style.SelectedValue == "Bold";
        Label1.Font.Italic = style.SelectedValue == "Italic";
    }

    protected void Page_Load(object sender, EventArgs e)
    {
        if (!IsPostBack)
            msg.Text = "First Page Load";
        else
            msg.Text = "Postback Occurred";
    }
}

using System;
using System.IO;

public partial class _Default : System.Web.UI.Page
{
    protected void btn_Click(object sender, EventArgs e)
    {
        string selected = "";

        foreach (System.Web.UI.WebControls.ListItem item in food.Items)
        {
            if (item.Selected)
                selected += item.Text + " ";
        }

        result.Text = "Welcome " + txtName.Text +
                      "<br/> Your Breakfast: " + selected;

        if (upload.HasFile)
        {
            string ext = Path.GetExtension(upload.FileName).ToLower();

            if (ext == ".jpg" || ext == ".jpeg" || ext == ".png")
            {
                string path = "~/Images/" + Path.GetFileName(upload.FileName);
                upload.SaveAs(Server.MapPath(path));
                img.ImageUrl = path;
            }
            else
            {
                result.Text += "<br/> Only JPG, JPEG and PNG allowed";
            }
        }
    }
}
