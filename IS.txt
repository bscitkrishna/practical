2. Count Vowels Using the Character Class

class Main {
    public static void main(String[] args) {
        String s = "Hello World";
        int count = 0;

        for (int i = 0; i < s.length(); i++) {
            char ch = Character.toLowerCase(s.charAt(i));

            if (ch == 'a' || ch == 'e' || ch == 'i' ||
                ch == 'o' || ch == 'u') {
                count++;
            }
        }

        System.out.println("Vowels: " + count);
    }
}

plo3 1. ListIterator — Print List Forward and Backward

import java.util.*;

class Main {
    public static void main(String[] args) {
        List<String> list = new ArrayList<>();
        list.add("Apple");
        list.add("Mango");
        list.add("Banana");

        ListIterator<String> it = list.listIterator();

        System.out.println("Forward:");
        while (it.hasNext())
            System.out.println(it.next());

        System.out.println("Backward:");
        while (it.hasPrevious())
            System.out.println(it.previous());
    }
}


2. Set and Iterator — Print Items Forward and Backward

A HashSet does not maintain order or support backward traversal directly. Use LinkedHashSet to preserve insertion order, then convert it to a list for reverse printing.

import java.util.*;

class Main {
    public static void main(String[] args) {
        Set<String> set = new LinkedHashSet<>();
        set.add("Apple");
        set.add("Mango");
        set.add("Banana");

        Iterator<String> it = set.iterator();

        System.out.println("Forward:");
        while (it.hasNext())
            System.out.println(it.next());

        List<String> list = new ArrayList<>(set);
        Collections.reverse(list);

        System.out.println("Backward:");
        for (String s : list)
            System.out.println(s);
    }
}

3. Set Interface — Add, Insert, Remove and Search Items

import java.util.*;

class Main {
    public static void main(String[] args) {
        Set<String> s1 = new HashSet<>();

        // a. Add items
        s1.add("Apple");
        s1.add("Mango");
        s1.add("Banana");
        System.out.println("Set: " + s1);

        // b. Insert items of one set into another
        Set<String> s2 = new HashSet<>();
        s2.add("Orange");
        s2.add("Grapes");
        s1.addAll(s2);
        System.out.println("After adding another set: " + s1);

        // c. Remove an item
        s1.remove("Mango");
        System.out.println("After removal: " + s1);

        // d. Search an item
        System.out.println("Apple found: " + s1.contains("Apple"));
    }
}

plo4 1. Map Interface Operations

import java.util.*;

class Main {
    public static void main(String[] args) {
        Map<Integer, String> m1 = new HashMap<>();

        // a. Add items
        m1.put(1, "Apple");
        m1.put(2, "Mango");
        m1.put(3, "Banana");

        // b. Remove item
        m1.remove(3);

        // c. Search key
        System.out.println("Key exists: " + m1.containsKey(1));

        // d. Get value
        System.out.println("Value: " + m1.get(1));

        // e. Insert one map into another
        Map<Integer, String> m2 = new HashMap<>();
        m2.put(4, "Orange");
        m2.put(5, "Grapes");
        m1.putAll(m2);

        // f. Print all keys and values
        for (Map.Entry<Integer, String> e : m1.entrySet()) {
            System.out.println(e.getKey() + " = " + e.getValue());
        }
    }
}
